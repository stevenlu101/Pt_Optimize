using System.Reflection;
using PtOptimize.Core;

var fl = BindingFlags.Static | BindingFlags.NonPublic | BindingFlags.Public;
var mDesign = tA.GetMethod("Design", fl); var mSet = tA.GetMethod("SetRW", fl); var mBuild = tA.GetMethod("Build", fl);
double I = 1214.0, T = 1150.0; double rho = Materials.PtResistivity(T) * 1e3;
var p = new DesignInputs();
string mode = args.Length > 0 ? args[0] : "all";
foreach (double R in new[] { 31.00, 30.50 })
foreach (double fine in new[] { 1.0, 0.0 })
{
    var d = (DesignSpec)mDesign.Invoke(null, new object[] { "W08" });
    d = d.Clone(); mSet.Invoke(null, new object[] { d, R, 30.0 });
    var tup = mBuild.Invoke(null, new object[] { d, p, fine, 0, null });
    var fields = tup.GetType().GetFields();
    var lc = (LineCase)fields[0].GetValue(tup); var m = (ShellMesh)fields[2].GetValue(tup);
    var on = ShellCurrent.Solve(m, I, rho, T);
    double pNet = I * I * rho / on.CurrentInA;
    Console.WriteLine($"== R{R} fine{fine} cells {m.CellCount} conv {on.Converged} it {on.Iterations} closure {on.FaceGenTotalW / pNet - 1:E2}");
    // (1) unconverged CG
    foreach (int it in new[] { 5, 30, 100 })
    {
        var u = ShellCurrent.Solve(m, I, rho, T, null, it);
        double pn = I * I * rho / u.CurrentInA;
        double vMax = u.V.Max(v => Math.Abs(v));
        Console.WriteLine($"  maxIter {it}: conv {u.Converged} Residual {u.Residual:E2} ResidAbsSum/Iin {u.ResidualAbsSum / u.CurrentInA:E2} closure {u.FaceGenTotalW / pn - 1:E2}  P_net {pn:0.000} (conv {pNet:0.000})  TotalGen/Pnet-1 {u.TotalGenW/pn-1:E2}");
    }
    // (2) allocation variants
    var clamp = m.ClampFaceActive ? m.ClampSetCells() : null;
    double scale = I / on.CurrentInA; double k2 = scale * scale * rho;
    int n = m.CellCount;
    var qH = new double[n]; var qS = new double[n]; var qD = new double[n];
    for (int k = 0; k < m.Faces.Count; k++)

        var f = m.Faces[k];
        if (f.B < 0)
        {
            if (f.Tag != ShellMesh.TagHole) continue;
            if (clamp != null && clamp[f.A]) continue;
            double gb = m.Thickness[f.A] * f.Length / f.DistAB; double va = on.V[f.A];
            double qq = gb * va * va * k2; qH[f.A] += qq; qS[f.A] += qq; qD[f.A] += qq; continue;
        }
        if (f.DistAB < 1e-12) continue;
        if (clamp != null && ShellMesh.IsClampFace(f, clamp, out int fc))
        {

            double dvc = on.V[f.A] - on.V[f.B]; double qq = gcl * dvc * dvc * k2;
            qH[fc] += qq; qS[fc] += qq; qD[fc] += qq; continue;
        }
        double sA = m.Thickness[f.A], sB = m.Thickness[f.B];
        double g = 2 * sA * sB / (sA + sB) * f.Length / f.DistAB;
        double dv = on.V[f.A] - on.V[f.B]; double qf = g * dv * dv * k2;
        qH[f.A] += 0.5 * qf; qH[f.B] += 0.5 * qf;
        double shA = sB / (sA + sB); qS[f.A] += shA * qf; qS[f.B] += (1 - shA) * qf;
        double rA = m.CentroidToFaceMm(f.A, f) / sA, rB = m.CentroidToFaceMm(f.B, f) / sB; double dA = rA / (rA + rB);
        qD[f.A] += dA * qf; qD[f.B] += (1 - dA) * qf;
    }
    double maxDiffH = 0; for (int i = 0; i < n; i++) maxDiffH = Math.Max(maxDiffH, Math.Abs(qH[i] - on.FaceGenW[i]) / Math.Max(1e-12, on.FaceGenW.Max()));

    double[] J(double[] q) { var j = new double[n]; for (int i = 0; i < n; i++) { double den = rho * m.Thickness[i] * m.Area[i]; j[i] = q[i] > 0 && den > 0 ? Math.Sqrt(q[i] / den) : 0; } return j; }

    double MaxRatio(double[] jj) { double r = 0; for (int i = 0; i < n; i++) if (on.JMagAPerMm2[i] >= 0.05 * on.JMaxAPerMm2) r = Math.Max(r, Math.Abs(jj[i] / on.JMagAPerMm2[i] - 1)); return r; }
    double L1(double[] a, double[] b) { double s = 0; for (int i = 0; i < n; i++) s += Math.Abs(a[i] - b[i]); return s; }
