using System;
using System.Diagnostics;
using PtOptimize.Core;
using Xunit;
using Xunit.Abstractions;
namespace PtOptimize.Tests;
public class ZZG1CovProbe
{
    private readonly ITestOutputHelper _o;
    public ZZG1CovProbe(ITestOutputHelper o) => _o = o;
    [Fact]
    public void 栅格步对拍两例的孔弧覆盖()
    {
        for (int which = 0; which < 2; which++)
        {
            var (g, lc, _, _, _, _, _, name) = DrawingPathParityTests.Case(which);
            double hF = lc.MeshFineMm, hC = lc.MeshCoarseMm, rF = lc.MeshFineRadiusMm, cl = lc.Base.BusbarClampLengthMm;
            var mA = FlangeMesher.Build(g, 0, hF, hC, rF, cl, lc.MeshInnerMm, lc.MeshInnerRadiusMm);
            _o.WriteLine($"{name} 生产 Build：覆盖率 {mA.HoleArcCoverage:R} 段数 {mA.HoleArcGapCount}");
            foreach (double s in new[] { 1.0, FlangeMesher.RasterStepFor(hF), 0.1 })
            {
                var f = AnalyticSurrogate.Rasterize(g, s, 2.0);
                var sw = Stopwatch.StartNew();
                var mF = FlangeMesher.BuildFromField(f, g.HoleRadiusMm, 0, hF, hC, rF, cl, lc.MeshInnerMm, lc.MeshInnerRadiusMm);
                double tb = sw.Elapsed.TotalMilliseconds; sw.Restart();
                double rD = PlateShapeAnalyzer.HoleRadiusOf(f);
                double tr = sw.Elapsed.TotalMilliseconds; sw.Restart();
                var cov = ShellMesh.MeasureHoleArcCoverage(mF);
                double tc = sw.Elapsed.TotalMilliseconds;
                _o.WriteLine($"{name} 栅格步 {s}（节点 {f.Nx}×{f.Nz}）：覆盖率 {mF.HoleArcCoverage:0.000000} 合计缺 {(1 - mF.HoleArcCoverage) * 2 * Math.PI * mF.HoleRadiusMm:0.000} mm 最长 {mF.HoleArcMaxGapMm:0.000} mm @ {mF.HoleArcMaxGapMidDeg:0.0}° 段数 {mF.HoleArcGapCount}　图纸孔半径 {rD:0.0000}（rh {g.HoleRadiusMm}）"
                           + $"　耗时：建网格 {tb:0} ms、读孔径 {tr:0.0} ms、量覆盖 {tc:0.00} ms");
            }
        }
    }
}
