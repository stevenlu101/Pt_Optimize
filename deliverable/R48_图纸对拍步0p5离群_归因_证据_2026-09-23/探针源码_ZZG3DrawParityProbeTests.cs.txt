    internal sealed class Sol
    {
        public ShellMesh M = null!;
        public ShellCurrentResult C = null!;
        public ShellThermalResult T = null!;
        public double TRoot;
        public PtProps Props = null!;
        public double[] FaceQ = Array.Empty<double>();   // 热：每条孔面 q = g·(T管 − T格)，非孔面 NaN
        public double[] FaceG = Array.Empty<double>();   // 热：每条孔面 g
        public double[] FaceGe = Array.Empty<double>();  // 电：每条孔面 t·L/DistAB（几何导度，mm）
    }

    internal static ShellMesh BuildMesh(FlangePlate g, LineCase lc, MeshRules rules, double rasterStep, IMaterialField? fieldOverride = null)
    {
        double hF = lc.MeshFineMm, hC = lc.MeshCoarseMm, rF = lc.MeshFineRadiusMm, cl = lc.Base.BusbarClampLengthMm;
        if (fieldOverride != null)
            return FlangeMesher.BuildFromMaterialWith(fieldOverride, g.HoleRadiusMm, rules, 0, hF, hC, rF, cl, lc.MeshInnerMm, lc.MeshInnerRadiusMm);
        if (rasterStep <= 0)
            return FlangeMesher.BuildWith(g, rules, null, 0, hF, hC, rF, cl, lc.MeshInnerMm, lc.MeshInnerRadiusMm);
        var f = AnalyticSurrogate.Rasterize(g, rasterStep, 2.0);
        return FlangeMesher.BuildFromMaterialWith(f, g.HoleRadiusMm, rules, 0, hF, hC, rF, cl, lc.MeshInnerMm, lc.MeshInnerRadiusMm);
    }

    internal static Sol SolveOn(ShellMesh m, LineCase lc, FlangePlate g, double iA, double tRoot, double tSet, double clampC, double tabInsul, bool anchor)
    {
        var p2 = SegmentSolver.Clone(lc.Base); p2.TSetC = tSet; p2.BusbarClampTempC = clampC;
        var sc = ShellCurrent.SolveFor(lc, m, iA, Materials.PtResistivity(tSet) * 1e3, tSet);
        Assert.True(sc.Converged);
        var th = ShellThermal.Solve(m, sc.JMagAPerMm2, p2, tRoot, g.InsulBoundaryXResolved, false,
                                    tabBoundaryX: g.Tangent().X, tabInsulThickMm: tabInsul, holeAnchorOnCircle: anchor);
        Assert.True(th.Converged);
        var s = new Sol { M = m, C = sc, T = th, TRoot = tRoot, Props = PtProps.For(p2) };
        int nf = m.Faces.Count;
        s.FaceQ = new double[nf]; s.FaceG = new double[nf]; s.FaceGe = new double[nf];
        for (int k = 0; k < nf; k++)
        {
            var f = m.Faces[k];
            if (f.B >= 0 || f.Tag != ShellMesh.TagHole) { s.FaceQ[k] = double.NaN; s.FaceG[k] = double.NaN; s.FaceGe[k] = double.NaN; continue; }
            double kA = s.Props.K(th.T[f.A]) * 1e-3 * m.Thickness[f.A];
            s.FaceG[k] = kA * f.Length / f.DistAB;
            s.FaceQ[k] = s.FaceG[k] * (tRoot - th.T[f.A]);
            s.FaceGe[k] = m.Thickness[f.A] * f.Length / f.DistAB;
        }
        return s;
    }

    static double Rc(ShellMesh m, int c) { var p = m.Centroid[c]; return Math.Sqrt(p.X * p.X + p.Z * p.Z); }
    static double Deg(Vec3 v) => Math.Atan2(v.Z, v.X) * 180 / Math.PI;

    internal static string Summary(Sol s, double rh)
    {
        var m = s.M;
        var hole = Enumerable.Range(0, m.Faces.Count).Where(k => m.Faces[k].B < 0 && m.Faces[k].Tag == ShellMesh.TagHole).ToList();
        var arc = hole.Where(k => !double.IsNaN(m.Faces[k].ArcRadiusMm)).ToList();
        var dn = arc.Select(k => Rc(m, m.Faces[k].A) - rh).ToArray();   // 有符号法向距（负 = 形心在孔内）
        var lens = arc.Select(k => m.Faces[k].Length).OrderBy(v => v).ToArray();
        var dist = hole.Select(k => m.Faces[k].DistAB).OrderBy(v => v).ToArray();
        int nb(Func<double, bool> p) => arc.Count(k => p(m.Faces[k].DistAB));
        var holeCells = hole.Select(k => m.Faces[k].A).Distinct().ToList();
        double jHole = holeCells.Count == 0 ? double.NaN : holeCells.Max(c => s.C.JMagAPerMm2[c]);
        int jHoleCell = holeCells.Count == 0 ? -1 : holeCells.OrderByDescending(c => s.C.JMagAPerMm2[c]).First();
        double sumG = hole.Sum(k => s.FaceG[k]), sumGe = hole.Sum(k => s.FaceGe[k]);
        double qSum = hole.Sum(k => s.FaceQ[k]);
        string med(double[] a) => a.Length == 0 ? "NaN" : a[a.Length / 2].ToString("0.0000");
        return $"格 {m.CellCount} 节点 {m.Nodes.Count} | 发热 {s.T.QGenW:0.000} W 抽热 {s.T.QFromTubeW:0.000} W（面和 {qSum:0.000}） 电发热 {s.C.TotalGenW:0.000} | "
             + $"孔面 {hole.Count}（弧 {arc.Count}，直边 {hole.Count - arc.Count}） 孔格 {holeCells.Count} | 弧长 min/med/max {lens.DefaultIfEmpty(double.NaN).Min():0.0000}/{med(lens)}/{lens.DefaultIfEmpty(double.NaN).Max():0.0000} Σ {lens.Sum():0.0000}（2πrh {2 * Math.PI * rh:0.0000}）| "
             + $"DistAB min/med/max {dist.DefaultIfEmpty(double.NaN).Min():0.00000}/{med(dist)}/{dist.DefaultIfEmpty(double.NaN).Max():0.0000} | 法向距分档 <0.01:{nb(d => d < 0.01)} <0.05:{nb(d => d < 0.05)} <0.1:{nb(d => d < 0.1)} <0.25:{nb(d => d < 0.25)} | "
             + $"有符号法向距 min {dn.DefaultIfEmpty(double.NaN).Min():+0.0000;-0.0000} | 形心在孔内 {m.HoleArcCentroidInside} 触下限 {m.HoleArcDistAtFloor} | 规则A并格 {m.HoleSliversMerged} 规则B并格 {m.SliversMerged} | 覆盖率 {m.HoleArcCoverage:0.000000} | "
             + $"Σg热 {sumG:0.0000} W/K Σ t·L/d {sumGe:0.000} mm | JMax {s.C.JMaxAPerMm2:0.000}（格 {s.C.JMaxCell} @ ({m.Centroid[Math.Max(0, s.C.JMaxCell)].X:0.00},{m.Centroid[Math.Max(0, s.C.JMaxCell)].Z:0.00})） 孔边 J 峰 {jHole:0.000}（@ ({(jHoleCell >= 0 ? m.Centroid[jHoleCell].X : double.NaN):0.00},{(jHoleCell >= 0 ? m.Centroid[jHoleCell].Z : double.NaN):0.00})） | 最热 {s.T.TMaxC:0.00} °C";
    }

/// <summary>混合材料源：矩形／线段离管轴最近点 &lt; rIn 的用 inner，否则用 outer。</summary>
internal sealed class G3HybridMaterial : IMaterialField
{
    readonly IMaterialField _in, _out; readonly double _r;
    public G3HybridMaterial(IMaterialField inner, IMaterialField outer, double rIn) { _in = inner; _out = outer; _r = rIn; }
    static double Near(double x0, double x1, double z0, double z1) { double nx = Math.Clamp(0, x0, x1), nz = Math.Clamp(0, z0, z1); return Math.Sqrt(nx * nx + nz * nz); }
    public (double XMin, double XMax, double ZMin, double ZMax, bool Exact) MaterialEnvelope() => _out.MaterialEnvelope();
    public MaterialIntegral Integrate(double x0, double x1, double z0, double z1) => (Near(x0, x1, z0, z1) < _r ? _in : _out).Integrate(x0, x1, z0, z1);
    public (double Length, double Mid) SegmentMaterial(bool vertical, double line, double a, double b)
    {
        double lo = Math.Min(a, b), hi = Math.Max(a, b);
        double nr = vertical ? Near(line, line, lo, hi) : Near(lo, hi, line, line);
        return (nr < _r ? _in : _out).SegmentMaterial(vertical, line, a, b);
    }
    public double FractionInsideCircle(double x0, double x1, double z0, double z1, double radiusMm) => (Near(x0, x1, z0, z1) < _r ? _in : _out).FractionInsideCircle(x0, x1, z0, z1, radiusMm);
    public string Describe() => $"混合（r<{_r} 用 {_in.Describe()}，其余 {_out.Describe()}）";
}

    static ThicknessField RasterShift(FlangePlate g, double step, double ox, double oz, double margin = 2.0)
    {
        var f0 = FlangeMesher.Rasterize(g, step, margin);
        // 平移为 0 的那一向照抄原栅格的原点与格数（坐标逐位同 Rasterize，免得浮点尾巴让直边 ±w 上的节点判内判外翻面）
        int nx = ox == 0 ? f0.Nx : f0.Nx + 1, nz = oz == 0 ? f0.Nz : f0.Nz + 1;
        double x0 = ox == 0 ? f0.X0 : f0.X0 - step + ox, z0 = oz == 0 ? f0.Z0 : f0.Z0 - step + oz;
        var f = new ThicknessField { X0 = x0, Z0 = z0, Step = step, Nx = nx, Nz = nz, T = new double[nx * nz],
            XMinMaterial = f0.XMinMaterial, XMaxMaterial = f0.XMaxMaterial, ZMinMaterial = f0.ZMinMaterial, ZMaxMaterial = f0.ZMaxMaterial };
        for (int i = 0; i < nx; i++)
            for (int j = 0; j < nz; j++)
            {
                double x = x0 + i * step, z = z0 + j * step;
                f.T[i * nz + j] = g.Inside(x, z) ? g.ThicknessAt(x, z) : 0.0;
            }
        return f;
    }

    static string Key(ShellMesh m, int c)

        var (x0, x1, z0, z1) = FlangeMesher.CellRect(m, c);
        return $"{x0:0.000000},{x1:0.000000},{z0:0.000000},{z1:0.000000}";

    static SortedSet<string> Merges(ShellMesh m)
    {
        var s = new SortedSet<string>();
        if (m.CellRects == null) return s;
        foreach (var rl in m.CellRects)
            if (rl != null && rl.Count > 1) s.Add(R(rl[0]) + " ← " + string.Join(" + ", rl.Skip(1).Select(R)));
        return s;
    }

    static ThicknessField Shift(FlangePlate g, double step, double ox, double oz)
    {
        var f0 = FlangeMesher.Rasterize(g, step, 2.0);
        // 平移为 0 的那一向照抄原栅格的原点与格数（坐标逐位同 Rasterize，免得浮点尾巴让直边 ±w 上的节点判内判外翻面）
        int nx = ox == 0 ? f0.Nx : f0.Nx + 1, nz = oz == 0 ? f0.Nz : f0.Nz + 1;
        double x0 = ox == 0 ? f0.X0 : f0.X0 - step + ox, z0 = oz == 0 ? f0.Z0 : f0.Z0 - step + oz;
        var f = new ThicknessField { X0 = x0, Z0 = z0, Step = step, Nx = nx, Nz = nz, T = new double[nx * nz],
            XMinMaterial = f0.XMinMaterial, XMaxMaterial = f0.XMaxMaterial, ZMinMaterial = f0.ZMinMaterial, ZMaxMaterial = f0.ZMaxMaterial };
        for (int i = 0; i < nx; i++)
            for (int j = 0; j < nz; j++)
            {
                double x = x0 + i * step, z = z0 + j * step;
                f.T[i * nz + j] = g.Inside(x, z) ? g.ThicknessAt(x, z) : 0.0;
            }
        return f;
    }

/// <summary>
/// G3 原型（只在探针里）：栅格场在孔圆 ±band 带内按解析圆判料（r ≥ rh 有料、r &lt; rh 无料），厚度取同一射线上 r = max(r, rh + s/2) 处的栅格值；带外照栅格。
/// 带内用 0.005 mm 子采样积分（近似，不是精确积分）。band = 一个栅格步（与 §0.-20 孔径失配的尺同一个）。
/// </summary>
internal sealed class G3CircleSnapField : IMaterialField
{
    readonly ThicknessField _f; readonly double _rh, _b, _sub; readonly bool _keepRasterT;
    public G3CircleSnapField(ThicknessField f, double rh, double band, double sub = 0.005, bool keepRasterT = false) { _f = f; _rh = rh; _b = band; _sub = sub; _keepRasterT = keepRasterT; }
    bool Touch(double x0, double x1, double z0, double z1)
    {
        double nx = Math.Clamp(0, x0, x1), nz = Math.Clamp(0, z0, z1);
        double near = Math.Sqrt(nx * nx + nz * nz);
        double fx = Math.Max(Math.Abs(x0), Math.Abs(x1)), fz = Math.Max(Math.Abs(z0), Math.Abs(z1));
        double far = Math.Sqrt(fx * fx + fz * fz);
        return near < _rh + _b && far > _rh - _b;
    }
    double T(double x, double z)
    {
        double r = Math.Sqrt(x * x + z * z);
        if (Math.Abs(r - _rh) >= _b) return _f.At(x, z);
        if (r < _rh) return 0;
        if (_keepRasterT)
        {
            double t0 = _f.At(x, z);
            if (t0 > 1e-9) return t0;
            double r1 = r + _f.Step;   // 栅格这点没料：沿射线往外一个栅格步取
            return r > 1e-12 ? _f.At(x * r1 / r, z * r1 / r) : 0;
        }
        double rs = Math.Max(r, _rh + 0.5 * _f.Step);
        return r > 1e-12 ? _f.At(x * rs / r, z * rs / r) : 0;
    }
    public (double XMin, double XMax, double ZMin, double ZMax, bool Exact) MaterialEnvelope() => _f.MaterialEnvelope();
    public MaterialIntegral Integrate(double x0, double x1, double z0, double z1)
    {
        if (!Touch(x0, x1, z0, z1)) return _f.Integrate(x0, x1, z0, z1);
        int nx = Math.Max(1, (int)Math.Ceiling((x1 - x0) / _sub)), nz = Math.Max(1, (int)Math.Ceiling((z1 - z0) / _sub));
        double dx = (x1 - x0) / nx, dz = (z1 - z0) / nz, a = dx * dz, A = 0, V = 0, Mx = 0, Mz = 0;
        for (int i = 0; i < nx; i++)
        {
            double x = x0 + (i + 0.5) * dx;
            for (int j = 0; j < nz; j++)
            {
                double z = z0 + (j + 0.5) * dz;
                double t = T(x, z);
                if (t <= 1e-9) continue;
                A += a; V += a * t; Mx += a * x; Mz += a * z;
            }
        }
        return new MaterialIntegral(A, V, Mx, Mz);
    }
    public (double Length, double Mid) SegmentMaterial(bool vertical, double line, double a, double b)
    {
        double lo = Math.Min(a, b), hi = Math.Max(a, b);
        bool touch = vertical ? Touch(line, line, lo, hi) : Touch(lo, hi, line, line);
        if (!touch) return _f.SegmentMaterial(vertical, line, a, b);
        int n = Math.Max(1, (int)Math.Ceiling((hi - lo) / _sub));
        double d = (hi - lo) / n, len = 0, mom = 0;
        for (int i = 0; i < n; i++)
        {
            double u = lo + (i + 0.5) * d;
            double t = vertical ? T(line, u) : T(u, line);
            if (t <= 1e-9) continue;
            len += d; mom += d * u;
        }
        return (len, len > 0 ? mom / len : 0.5 * (lo + hi));
    }
    public double FractionInsideCircle(double x0, double x1, double z0, double z1, double radiusMm) => _f.FractionInsideCircle(x0, x1, z0, z1, radiusMm);
    public string Describe() => $"孔圆带内按解析圆判料（原型，带 ±{_b}）+ {_f.Describe()}";
}

                var (g, lc, iA, tRoot, tSet, clampC, tabInsul, name) = DrawingPathParityTests.Case(which);
                // 方法 1：生产栅格步，倍率 1.0／0.5
                foreach (double sc in new[] { 1.0, 0.5 })
                {
                    double hF = lc.MeshFineMm * sc, hC = lc.MeshCoarseMm * sc;
                    var mA = FlangeMesher.Build(g, 0, hF, hC, lc.MeshFineRadiusMm, lc.Base.BusbarClampLengthMm, lc.MeshInnerMm, lc.MeshInnerRadiusMm);
                    var a = DrawingPathParityTests.SolveOne(mA, lc, g, iA, tRoot, tSet, clampC, tabInsul);
                    double step = FlangeMesher.RasterStepFor(hF);
                    var b = DrawingPathParityTests.SolveOne(MkF(g, lc, step, hF, hC), lc, g, iA, tRoot, tSet, clampC, tabInsul);
                    double dG = Math.Abs(b.QGen - a.QGen) / a.QGen * 100, dT = b.QFromTube - a.QFromTube;
                    W($"  方法1 {name} 倍率 {sc} 步 {step}：发热差 {dG:0.000} % 抽热差 {dT:+0.000;-0.000} W ⇒ {(dG <= 0.1 && Math.Abs(dT) <= 1.0 ? "过" : "红")}");
                }
                {
                    double hF = lc.MeshFineMm, hC = lc.MeshCoarseMm;
                    var mA = FlangeMesher.Build(g, 0, hF, hC, lc.MeshFineRadiusMm, lc.Base.BusbarClampLengthMm, lc.MeshInnerMm, lc.MeshInnerRadiusMm);
                    var a = DrawingPathParityTests.SolveOne(mA, lc, g, iA, tRoot, tSet, clampC, tabInsul);
                    var d = new List<(double g, double t)>();
                    foreach (double st in new[] { 1.0, FlangeMesher.RasterStepFor(hF), 0.1 })
                    {
                        var b = DrawingPathParityTests.SolveOne(MkF(g, lc, st, hF, hC), lc, g, iA, tRoot, tSet, clampC, tabInsul);
                        d.Add((Math.Abs(b.QGen - a.QGen) / a.QGen * 100, b.QFromTube - a.QFromTube));
                    }
                    bool a1 = d[2].g <= 0.1, a2 = Math.Abs(d[2].t) <= 0.1, a3 = d[1].g <= 0.1 && Math.Abs(d[1].t) <= 1.0,
                         a4 = Math.Abs(d[0].t) >= Math.Abs(d[1].t) - 1e-9 && Math.Abs(d[1].t) >= Math.Abs(d[2].t) - 1e-9;
                    W($"  方法2 {name}：抽热差 1.0 {d[0].t:+0.000;-0.000} → 0.5 {d[1].t:+0.000;-0.000} → 0.1 {d[2].t:+0.000;-0.000} W；发热差 {d[0].g:0.000}/{d[1].g:0.000}/{d[2].g:0.000} %；断言 步0.1发热 {a1} 步0.1抽热 {a2} 生产步 {a3} 单调 {a4} ⇒ {(a1 && a2 && a3 && a4 ? "过" : "红")}");
                }
            }
        }
