# 决 103：HANDOVER §1.83 判据表应改成的行（生成器）。用法：python3 patch183.py HANDOVER.md 输出路径  —— 只改表里 10 行、插 3 行，其余逐字不动。
import sys
src, dst = sys.argv[1], sys.argv[2]
raw = open(src, 'rb').read(); bom = raw.startswith(b'\xef\xbb\xbf')
lines = raw.decode('utf-8-sig').split('\n')
start = next(i for i, l in enumerate(lines) if l.startswith('## 1.83'))
end = next(i for i in range(start + 1, len(lines)) if lines[i].startswith('### ') or lines[i].startswith('## '))
def cells(l):
    assert l.startswith('|') and l.rstrip().endswith('|'), l[:40]
    return [c for c in l.strip()[1:-1].split('|')]
def join(cs): return '|' + '|'.join(cs) + '|'
def find(prefix):
    hit = [i for i in range(start, end) if lines[i].startswith('| `' + prefix)]
    assert len(hit) == 1, (prefix, hit)
    return hit[0]
D = '**决 103（业主 2026-09-24）**'
# ① 升温：限值 12 → 11
i = find('① 升温'); c = cells(lines[i])
assert c[1].strip() == '12 A/mm²'; c[1] = ' 11 A/mm² '
c[3] = f' {D}：限值 = 卡交付的管 J 限值 = min(参数表「管许用电流密度」12, 「管 J 使用上限」11) = 11（业主原话「以管的最大使用电流密度 J < 11 与 20 °C/h 为限」）；原 12 ⇒ 11，变因同「管 J」行。' + c[3]
lines[i] = join(c)
# ②′ ⑦ ⑧：降为参考
for p, why in [('②′管孔净流入', '管孔净热流的合格方向反向（原「须为正」⇒ 现行冷侧「管接触处流入法兰的净热流 ≤ 0」）'),
               ('⑦ 最热铂高出热偶读数', '基准改为模型算的管接触处温度、±5 作废（现行热侧「法兰最热处高出管接触处温度 ≤ 10」）'),
               ('⑧ 管根低于热偶读数', '拉低管温的方向改由「管接触处流入法兰的净热流 ≤ 0」卡、±5 作废')]:
    i = find(p); c = cells(lines[i])
    c[2] = ' 参考（2026-09-24 起照印不判） '
    c[3] = f' {D}：降为参考 —— {why}。以下为原出处：' + c[3]
    c[4] = ' 参考 '
    lines[i] = join(c)
# 新两行插在 ⑧ 之后
i8 = find('⑧ 管根低于热偶读数')
new_hot = ('| `法兰最热处高出管接触处温度`（第 j 片 max(该片法兰温度场) − T_接触,j；T_接触,j = 模型算的该片管根接触温度，端片取相邻段端温、共用片取两侧段端温度的较高者，与旧判法 ②″ 同一基准） '
           '| 10 K | **硬判据** | ' + D + '：「稳态期法兰温度场比铂金管接触处温度高的方向是对的、可容许到 10 °C 以内，拉低管温的方向是错的，之前的 ±5 作废」；'
           '与 08-25 口述链「法兰可高于管 10 °C 以内」一致。限值 = 参数表「法兰最热处高出管接触处温度 允许值」（`DesignInputs.HotOverContactAllowK`，默认 `CriteriaRules.HotOverContactMaxKDefault`）；'
           '热偶设定值只印对照。改回参数 `DesignInputs.CriteriaRuleSet = 决103前` 时不出这一行 '
           '| 参考（全局方案第 2 版 1.3 节：空管到温稳态只卡电流密度与几何闭式） |')
new_cold = ('| `管接触处流入法兰的净热流`（第 j 片管孔处由管流入法兰的净热流 = FlangeOut.QFromTubeW，正 = 法兰在抽管子的热、把管在接触处拉低）须 ≤ 0 '
            '| 0 W | **硬判据** | ' + D + '：「拉低管温的方向是错的」⇒ 法兰不得拉低管在接触处的温度 ⇔ 管孔净热流不得由管流入法兰；**不给预算**（限值 `CriteriaRules.TubeToFlangeHeatMaxW` = 0）。'
            '与「管孔净流入」同一个量、合格方向相反。改回参数下不出这一行 '
            '| 参考（同上） |')
lines[i8 + 1:i8 + 1] = [new_hot, new_cold]
# 管 J：12 → 11；对照行插在后面
i = find('管 J'); c = cells(lines[i])
assert c[1].strip() == '12 A/mm²'; c[1] = ' 11 A/mm² '
c[3] = (f' {D}：「以管的最大使用电流密度 J < 11 与 20 °C/h（可控硅）为限」⇒ 卡交付的限值 = min(参数表「管许用电流密度」12, 「管 J 使用上限」11)'
        '（`DesignInputs.TubeJLimitAPerMm2`，唯一读法；升温 ①、升温全程逐点、升温电流上限读同一个数）；比较符沿用「≤」（原话「< 11」，恰等于 11 测度为零）；'
        '原 12 那一行留作对照（下一行）；变因：12 ⇒ 11。以下为原出处：' + c[3])
lines[i] = join(c)
iJ = find('管 J')
legacy = ('| `· 管电流密度对原许用值（对照）` | 12 A/mm² | 参考 | ' + D + ' 前卡交付的管 J 限值（用户 2026-08-15 现场：一般 15；管壁 0.6 时 12 是极限；参数表「管许用电流密度」`TubeJAllowAPerMm2` 默认 12）。'
          '2026-09-24 起卡交付的换成上一行（与使用上限 11 取小），本行照印作对照 | 参考 |')
lines[iJ + 1:iJ + 1] = [legacy]
# 两条热稳定：参考 → 硬判据（带玻璃稳态）
for p, extra in [('· 整片热稳定', '逐片评、取最小（FlangeLumped）'), ('· 局部热稳定', '**全格精算**（决 99 选项 A：全片自由格逐格同一个 LocalStability.Check；原前 12 名口径在 W08／W06 上乐观 3.5～4.9 倍、落点偏近 r 31.5 → 真 r 66），片上落点接 C4′ 细区半径放大')]:
    i = find(p); c = cells(lines[i])
    assert c[2].strip() == '参考', c[2]
    c[2] = ' **硬判据** '
    c[3] = (f' {D}：由参考升为带玻璃稳态硬判据（须 ≥ 1.0）—— 允许法兰偏热后，原「管孔净流入永远先红」的隐性保护消失（HO:9075、9260）；{extra}；'
            '升温全程逐点同样卡（全局方案第 2 版 1.2 节）。以下为原出处：' + c[3])
    c[4] = ' 参考（全局方案第 2 版 1.3 节：空管到温稳态只卡电流密度与几何闭式；升温期的热稳定由升温全程卡） '
    lines[i] = join(c)
out = '\n'.join(lines)
open(dst, 'wb').write((b'\xef\xbb\xbf' if bom else b'') + out.encode('utf-8'))
s2 = next(i for i, l in enumerate(lines) if l.startswith('## 1.83'))
e2 = next(i for i in range(s2 + 1, len(lines)) if lines[i].startswith('### ') or lines[i].startswith('## '))
tbl = [l for l in lines[s2:e2] if l.startswith('|')]
open(dst + '.table.md', 'w', encoding='utf-8').write('\n'.join(tbl) + '\n')
print('ok', len(tbl))
