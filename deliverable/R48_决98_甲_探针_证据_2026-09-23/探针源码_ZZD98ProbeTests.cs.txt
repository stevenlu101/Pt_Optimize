    /// <summary>孔空腔：照抄 PlateShapeAnalyzer.FindVoids（私有）—— 从图幅边界填充「外部空白」，剩下的无料连通域里最大的是管孔。返回孔空腔节点下标（ix*nz+iz）与外部标记。</summary>
    internal static (List<int> hole, bool[] outside, bool[] isHole) HoleVoid(ThicknessField f)
    {
        int nx = f.Nx, nz = f.Nz, n = nx * nz;
        var outside = new bool[n];
        var stack = new Stack<int>();
        void Push(int ix, int iz)
        {
            if (ix < 0 || ix >= nx || iz < 0 || iz >= nz) return;
            int id = ix * nz + iz;
            if (outside[id] || f.T[id] > 1e-6) return;
            outside[id] = true; stack.Push(id);
        }
        for (int ix = 0; ix < nx; ix++) { Push(ix, 0); Push(ix, nz - 1); }
        for (int iz = 0; iz < nz; iz++) { Push(0, iz); Push(nx - 1, iz); }
        while (stack.Count > 0) { int id = stack.Pop(); int ix = id / nz, iz = id % nz; Push(ix + 1, iz); Push(ix - 1, iz); Push(ix, iz + 1); Push(ix, iz - 1); }
        var seen = new bool[n]; List<int>? best = null;
        for (int id0 = 0; id0 < n; id0++)
        {
            if (f.T[id0] > 1e-6 || outside[id0] || seen[id0]) continue;
            var comp = new List<int>(); var st = new Stack<int>(); st.Push(id0); seen[id0] = true;
            while (st.Count > 0)
            {
                int id = st.Pop(); comp.Add(id); int ix = id / nz, iz = id % nz;
                foreach (var (dx, dz) in new[] { (1, 0), (-1, 0), (0, 1), (0, -1) })
                {
                    int jx = ix + dx, jz = iz + dz;
                    if (jx < 0 || jx >= nx || jz < 0 || jz >= nz) continue;
                    int jd = jx * nz + jz;
                    if (seen[jd] || outside[jd] || f.T[jd] > 1e-6) continue;
                    seen[jd] = true; st.Push(jd);
                }
            }
            if (best is null || comp.Count > best.Count) best = comp;
        }
        best ??= new List<int>();
        var isHole = new bool[n]; foreach (int id in best) isHole[id] = true;
        return (best, outside, isHole);
    }

    internal sealed record Est(double A, double B, double C1, double C2, double D, double DLo, double DHi,
                               int BViolRaster, int NHole, int NEdge, double BMinusA);

    /// <summary>
    /// 全部量法（圆心 = 管轴原点，与网格生成器同一个框；(c2) 另放开圆心）：
    ///  (a) HoleRadiusOf（生产，等面积）；(b) 环带 [rh − 2s, rh + 2s] 内有料面积 A（有料节点方格 ∩ 环带，闭式）⇒ √(R_out² − A/π)；
    ///  (c1) 孔空腔节点与四邻有料节点的连线中点，定心均值半径；(c2) 同一组中点的 Kåsa 代数圆拟合（圆心自由）；
    ///  (d) 分离区间：r_lo = 孔空腔节点的最大半径，r_hi = 有料节点的最小半径；点采样 + 圆心在原点的圆孔 ⇒ 真半径 ∈ (r_lo, r_hi]，取中点，半宽 = 自带误差界。
    /// </summary>
    internal static Est Measure(ThicknessField f, double rhDesign)
    {
        double s = f.Step, h = 0.5 * s;
        var (hole, outside, isHole) = HoleVoid(f);
        int nz = f.Nz;
        double a = Math.Sqrt(hole.Count * s * s / Math.PI);
        // (b)
        double rin = rhDesign - 2 * s, rout = rhDesign + 2 * s, A = 0; int viol = 0;
        int iLo = Math.Max(0, (int)Math.Floor((-rout - s - f.X0) / s)), iHi = Math.Min(f.Nx - 1, (int)Math.Ceiling((rout + s - f.X0) / s));
        int jLo = Math.Max(0, (int)Math.Floor((-rout - s - f.Z0) / s)), jHi = Math.Min(f.Nz - 1, (int)Math.Ceiling((rout + s - f.Z0) / s));
        for (int i = iLo; i <= iHi; i++)
            for (int j = jLo; j <= jHi; j++)
            {
                double cx = f.X0 + i * s, cz = f.Z0 + j * s;
                double x0 = cx - h, x1 = cx + h, z0 = cz - h, z1 = cz + h;
                double nx_ = Math.Clamp(0, x0, x1), nz_ = Math.Clamp(0, z0, z1);
                double near = Math.Sqrt(nx_ * nx_ + nz_ * nz_);
                double fx = Math.Max(Math.Abs(x0), Math.Abs(x1)), fz = Math.Max(Math.Abs(z0), Math.Abs(z1));
                double far = Math.Sqrt(fx * fx + fz * fz);
                int id = i * nz + j;
                bool mat = f.T[id] > 1e-6;
                bool touchAnn = near < rout && far > rin;
                if (mat && touchAnn)
                    A += R48HoleBandCircleGateTests.RectDiskArea(x0, x1, z0, z1, rout) - R48HoleBandCircleGateTests.RectDiskArea(x0, x1, z0, z1, rin);
                // 假设核实（栅格）：H1 碰环带的无料方格都属孔空腔；H2 碰圆盘 R_in 的方格都是孔空腔；H3 孔空腔方格都在 R_out 以内
                if (!mat && touchAnn && !isHole[id]) viol++;
                if (near < rin && !isHole[id]) viol++;
                if (isHole[id] && far > rout) viol++;
            }
        double b = Math.Sqrt(Math.Max(0, rout * rout - A / Math.PI));
        // (c)、(d)
        double sumR = 0; int ne = 0; double lo = double.NegativeInfinity, hi = double.PositiveInfinity;
        double Sxx = 0, Sxz = 0, Szz = 0, Sx = 0, Sz = 0, Sn = 0, Sxw = 0, Szw = 0, Sw = 0;   // Kåsa：min Σ(w + D x + E z + F)²，w = x² + z²
        foreach (int id in hole)
        {
            int ix = id / nz, iz = id % nz;
            double x = f.X0 + ix * s, z = f.Z0 + iz * s;
            lo = Math.Max(lo, Math.Sqrt(x * x + z * z));
            foreach (var (dx, dz) in new[] { (1, 0), (-1, 0), (0, 1), (0, -1) })
            {
                int jx = ix + dx, jz = iz + dz;
                if (jx < 0 || jx >= f.Nx || jz < 0 || jz >= nz) continue;
                int jd = jx * nz + jz;
                if (f.T[jd] <= 1e-6) continue;
                double mx = x + 0.5 * dx * s, mz = z + 0.5 * dz * s;
                sumR += Math.Sqrt(mx * mx + mz * mz); ne++;
                double w = mx * mx + mz * mz;
                Sxx += mx * mx; Sxz += mx * mz; Szz += mz * mz; Sx += mx; Sz += mz; Sn += 1; Sxw += mx * w; Szw += mz * w; Sw += w;
            }
        }
        // r_hi：孔附近（|p| < r_lo + 2s 的窗口里）全部有料节点的最小半径（窗口外的有料节点半径更大，不影响最小值）
        double win = lo + 2 * s;
        int i0 = Math.Max(0, (int)Math.Floor((-win - f.X0) / s)), i1 = Math.Min(f.Nx - 1, (int)Math.Ceiling((win - f.X0) / s));
        int j0 = Math.Max(0, (int)Math.Floor((-win - f.Z0) / s)), j1 = Math.Min(f.Nz - 1, (int)Math.Ceiling((win - f.Z0) / s));
        for (int i = i0; i <= i1; i++)
            for (int j = j0; j <= j1; j++)
            {
                if (f.T[i * nz + j] <= 1e-6) continue;
                double x = f.X0 + i * s, z = f.Z0 + j * s;
                hi = Math.Min(hi, Math.Sqrt(x * x + z * z));
            }
        double c1 = ne > 0 ? sumR / ne : double.NaN;
        // Kåsa 正规方程 [Sxx Sxz Sx; Sxz Szz Sz; Sx Sz Sn]·[D E F] = −[Sxw Szw Sw]
        double c2 = double.NaN;
        {
            var M = new double[3, 3] { { Sxx, Sxz, Sx }, { Sxz, Szz, Sz }, { Sx, Sz, Sn } };
            var rhs = new[] { -Sxw, -Szw, -Sw };
            var sol = Solve3(M, rhs);
            if (sol is not null) { double D = sol[0], E = sol[1], Fk = sol[2]; c2 = Math.Sqrt(D * D / 4 + E * E / 4 - Fk); }
        }
        double d = lo < hi ? 0.5 * (lo + hi) : double.NaN;
        return new Est(a, b, c1, c2, d, lo, hi, viol, hole.Count, ne, b - a);
    }

    static double[]? Solve3(double[,] M, double[] r)
    {
        double det(double[,] m) => m[0, 0] * (m[1, 1] * m[2, 2] - m[1, 2] * m[2, 1]) - m[0, 1] * (m[1, 0] * m[2, 2] - m[1, 2] * m[2, 0]) + m[0, 2] * (m[1, 0] * m[2, 1] - m[1, 1] * m[2, 0]);
        double D0 = det(M); if (Math.Abs(D0) < 1e-300) return null;
        var x = new double[3];
        for (int k = 0; k < 3; k++)
        {
            var Mk = (double[,])M.Clone();
            for (int i = 0; i < 3; i++) Mk[i, k] = r[i];
            x[k] = det(Mk) / D0;
        }
        return x;
    }

    /// <summary>孔周窗口栅格（只供量法的二维相位扫描）：节点 = (ox + i·s, oz + j·s)，窗口半宽 wHalf；点采样 g.Inside／ThicknessAt，与 Rasterize 同一式。</summary>
    static ThicknessField Window(FlangePlate g, double s, double ox, double oz, double wHalf)
    {
        int iA = (int)Math.Floor((-wHalf - ox) / s), iB = (int)Math.Ceiling((wHalf - ox) / s);
        int jA = (int)Math.Floor((-wHalf - oz) / s), jB = (int)Math.Ceiling((wHalf - oz) / s);
        int nx = iB - iA + 1, nz = jB - jA + 1;
        var f = new ThicknessField { X0 = ox + iA * s, Z0 = oz + jA * s, Step = s, Nx = nx, Nz = nz, T = new double[nx * nz] };
        for (int i = 0; i < nx; i++)
            for (int j = 0; j < nz; j++)
            {
                double x = ox + (iA + i) * s, z = oz + (jA + j) * s;
                f.T[i * nz + j] = g.Inside(x, z) ? g.ThicknessAt(x, z) : 0.0;
            }
        return f;
    }

    /// <summary>环带假设的解析核实：极坐标密采样（角 3600 × 径向 0.01 mm），环带 [rh − 2s, rh + 2s] 内 g.Inside 恰等于「r ≥ 孔半径」的点比例（1 = 环带里只有孔边，没有板缘、舌根、开槽）。</summary>
    static (int bad, int total) AnnulusAnalyticCheck(FlangePlate g, double rin, double rout)
    {
        int bad = 0, tot = 0; double R = g.HoleRadiusMm;
        for (int k = 0; k < 3600; k++)
        {
            double th = 2 * Math.PI * (k + 0.5) / 3600, c = Math.Cos(th), sn = Math.Sin(th);
            for (double r = rin + 0.005; r < rout; r += 0.01)
            {
                bool want = r >= R; bool got = g.Inside(r * c, r * sn);
                tot++; if (want != got) bad++;
            }
        }
        return (bad, tot);
    }

    static int ArcFaces(ShellMesh m) => m.Faces.Count(fc => fc.B < 0 && !double.IsNaN(fc.ArcRadiusMm));
    static double ArcR(ShellMesh m) { var a = m.Faces.Where(fc => fc.B < 0 && !double.IsNaN(fc.ArcRadiusMm)).Select(fc => fc.ArcRadiusMm).ToArray(); return a.Length == 0 ? double.NaN : a.Average(); }

    /// <summary>原型甲：带内按 rMat 判料（HoleBandCircleField 公开构造，带宽 1 × s），弧面、热锚点、孔标签按 rArc 建；锚点照 BuildFromField 从原始栅格推。</summary>
    internal static ShellMesh MeshJ(ThicknessField tf, double rArc, double rMat, LineCase lc)
    {
        var (xa, za) = FlangeMesher.AnchorsFromField(tf);
        var fld = new HoleBandCircleField(tf, rMat, FlangeMesher.HoleBandPerStep * tf.Step);
        var m = FlangeMesher.BuildFromMaterialWith(fld, rArc, MeshRules.Production, 0, lc.MeshFineMm, lc.MeshCoarseMm, lc.MeshFineRadiusMm, lc.Base.BusbarClampLengthMm,
                                                   lc.MeshInnerMm, lc.MeshInnerRadiusMm, false, xa, za, 0, clampBandMm: 0, clampFullFace: true, clampFaceDirichlet: true);
        m.SourceField = tf;
        return m;
    }

    static double Q(ShellMesh m, LineCase lc, FlangePlate g, double iA, double tRoot, double tSet, double clampC, double tabInsul, out double qGen)
    {
        try { var r = DrawingPathParityTests.SolveOne(m, lc, g, iA, tRoot, tSet, clampC, tabInsul); qGen = r.QGen; return r.QFromTube; }
        catch (Exception) { qGen = double.NaN; return double.NaN; }
    }
