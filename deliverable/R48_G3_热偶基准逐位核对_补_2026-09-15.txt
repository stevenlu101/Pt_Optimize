sealed class DirCtx : AssemblyLoadContext

    readonly string _dir;
    public DirCtx(string dir, string name) : base(name) { _dir = dir; }

    {

    public object MakeResult(double[] sp, (double disc, double tab)[] pl, bool flangesNull = false, bool segmentsNull = false)
    {
        var r = Activator.CreateInstance(_lr);
        var segs = Array.CreateInstance(_so, sp.Length);
        for (int i = 0; i < sp.Length; i++)
        {
            var s = Activator.CreateInstance(_so);
            _so.GetField("SetpointC").SetValue(s, sp[i]);
            segs.SetValue(s, i);
        }
        var fls = Array.CreateInstance(_fo, pl.Length);
        for (int j = 0; j < pl.Length; j++)
        {
            var f = Activator.CreateInstance(_fo);
            _fo.GetField("TDiscMaxC").SetValue(f, pl[j].disc);
            _fo.GetField("TTabMaxC").SetValue(f, pl[j].tab);
            fls.SetValue(f, j);
        }
        _lr.GetField("Segments").SetValue(r, segmentsNull ? null : segs);
        _lr.GetField("Flanges").SetValue(r, flangesNull ? null : fls);
        return r;
    }
    public double Hot(object r, int j) => (double)Inv(_hot, r, j);
