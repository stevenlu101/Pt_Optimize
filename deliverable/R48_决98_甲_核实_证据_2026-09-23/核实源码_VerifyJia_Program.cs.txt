    static ThicknessField Shift(FlangePlate g, double s, double ox, double oz)
    {
        var f0 = FlangeMesher.Rasterize(g, s, 2.0);
        if (ox == 0 && oz == 0) return f0;
        int nx = ox == 0 ? f0.Nx : f0.Nx + 1, nz = oz == 0 ? f0.Nz : f0.Nz + 1;
        double x0 = ox == 0 ? f0.X0 : f0.X0 - s + ox, z0 = oz == 0 ? f0.Z0 : f0.Z0 - s + oz;
        var f = new ThicknessField { X0 = x0, Z0 = z0, Step = s, Nx = nx, Nz = nz, T = new double[nx * nz] };
        for (int i = 0; i < nx; i++) for (int j = 0; j < nz; j++)
        { double x = x0 + i * s, z = z0 + j * s; f.T[i * nz + j] = g.Inside(x, z) ? g.ThicknessAt(x, z) : 0.0; }
        return f;
    }

    // 自写孔空腔：从离原点最近的无料节点出发的四连通无料域
    static bool[] HoleSet(ThicknessField f, out int n)
    {
        int nx = f.Nx, nz = f.Nz; var hole = new bool[nx * nz];
        int best = -1; double bd = double.MaxValue;
        for (int i = 0; i < nx; i++) for (int j = 0; j < nz; j++)
        { double x = f.X0 + i * f.Step, z = f.Z0 + j * f.Step; double d = x * x + z * z; if (d < bd && f.T[i * nz + j] <= 1e-6) { bd = d; best = i * nz + j; } }
        var q = new Queue<int>(); q.Enqueue(best); hole[best] = true; n = 0;
        while (q.Count > 0)
        {
            int id = q.Dequeue(); n++; int i = id / nz, j = id % nz;
            foreach (var (di, dj) in new[] { (1, 0), (-1, 0), (0, 1), (0, -1) })
            {
                int a = i + di, b = j + dj; if (a < 0 || b < 0 || a >= nx || b >= nz) continue;
                int k = a * nz + b; if (hole[k] || f.T[k] > 1e-6) continue; hole[k] = true; q.Enqueue(k);
            }
        }
        return hole;
    }

    // 方格 [x0,x1]×[z0,z1] 与圆盘 R 的面积：x 向 4000 段中点求积，z 向解析（独立于 RectDiskArea 闭式）
    static double SqDisk(double x0, double x1, double z0, double z1, double R)
    {
        int n = 4000; double dx = (x1 - x0) / n, A = 0;
        for (int k = 0; k < n; k++)
        {
            double x = x0 + (k + 0.5) * dx; if (Math.Abs(x) >= R) continue;
            double h = Math.Sqrt(R * R - x * x); double lo = Math.Max(z0, -h), hi = Math.Min(z1, h);
            if (hi > lo) A += (hi - lo) * dx;
        }
        return A;
    }

    record M(double A_prod, double A_mine, int Nprod, int Nmine, double B, double Lo, double Hi, double D, int holeCellsOutsideRout, int nonHoleVoidInAnn);

    static M Measure(ThicknessField f, double rhDesign)
    {
        double s = f.Step, hs = s / 2; int nz = f.Nz;
        double aProd = PlateShapeAnalyzer.HoleRadiusOf(f);
        int nProd = (int)Math.Round(aProd * aProd * Math.PI / (s * s));
        var hole = HoleSet(f, out int n);
        double aMine = Math.Sqrt(n * s * s / Math.PI);
        double rin = rhDesign - 2 * s, rout = rhDesign + 2 * s, A = 0; int v1 = 0, v2 = 0;
        double lo = double.NegativeInfinity, hi = double.PositiveInfinity;
        for (int i = 0; i < f.Nx; i++) for (int j = 0; j < nz; j++)
        {
            double x = f.X0 + i * s, z = f.Z0 + j * s; int id = i * nz + j; double r = Math.Sqrt(x * x + z * z);
            bool mat = f.T[id] > 1e-6;
            if (hole[id]) lo = Math.Max(lo, r);
            if (mat) hi = Math.Min(hi, r);
            if (r > rout + s || r < rin - s) { if (hole[id] && r + s > rout) v1++; continue; }
            double x0 = x - hs, x1 = x + hs, z0 = z - hs, z1 = z + hs;
            double far = Math.Sqrt(Math.Pow(Math.Max(Math.Abs(x0), Math.Abs(x1)), 2) + Math.Pow(Math.Max(Math.Abs(z0), Math.Abs(z1)), 2));
            double nx_ = Math.Clamp(0, x0, x1), nz_ = Math.Clamp(0, z0, z1); double near = Math.Sqrt(nx_ * nx_ + nz_ * nz_);
            if (mat) A += SqDisk(x0, x1, z0, z1, rout) - SqDisk(x0, x1, z0, z1, rin);
            if (hole[id] && far > rout) v1++;
            if (!mat && !hole[id] && near < rout && far > rin) v2++;
        }
        double b = Math.Sqrt(rout * rout - A / Math.PI);
        return new M(aProd, aMine, nProd, n, b, lo, hi, 0.5 * (lo + hi), v1, v2);
    }
